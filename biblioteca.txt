'''
Entidade: persistencia - representa como os dados são armazenados no banco 
          estão acopladas ao ORM (JPA/Hibernate, EF Core, TypeORM etc.)
          Tem anotações e ajustes de acordo com a biblioteca de banco que uso
          
Dominio:  model - representa o coração do negócio, contém regras, comportamentos
          e validações. Não conhece banco, API, framework
          
DTO :     Data Transfer Objects, representam entrada(request) e saídas (response) de serviços. 
          Transportar dados entre camadas, especialmente entre Controller ↔ Service
          São definir contratos da API
          
Controlador: Controller quando necessário, são responsáveis por request via http, ou seja, chamada remota
          mas não devem ter implementação de regras de negócio, apenas validam as informações
          nas DTOs de entrada e repassam para o Serviço que implementam essas regras. 
          Nunca chamar banco diretamente.
          
Serviço:  Orquestrar o fluxo do caso de uso, converter DTO ↔ Domínio ↔ Entidade
          Coordenar regras
          
Repositories: Responsabilidade de comunicação com o banco, persistência e consulta 
              retornar Entidades ou Domínio (dependendo da estratégia)
              
ESTRUTURA Comum de Pasta no projeto:
\---src
    +---app                     /* app.whl */
    +---core                    /* core.whl */
    |   +---domain
    |   +---entities
    |   \---repositories
    \---services                /* services.whl */
        +---bos
        \---dto
        
OBS.: Em Python, o equivalente prático de “projetos gerando DLLs”
      são packages distribuíveis (.whl), organizados em módulos 
      independentes, instalados via pip.
          

Sistema controla emprestimos, onde:
cadastra livors, 
cadastra usuários,
operacao: empresta, devolve, lista emprestimos
Livro(titulo, autor, isbn)
Usuario(nome, email) empresta livros, 
Emprestimo(Usuario, Livro)
'''

class Livro:
   def __init__(self,titulo, autor, isbn):
    self.titulo = titulo.strip()
    self.autor = autor.strip()
    self.isbn = isbn.strip()



class Usuario:
   def __init__(self, nome, email):
     self.nome = nome.strip()
     self.email = email.strip()



class Emprestimo:
   def __init__(self, usuario, livro):
     self.usuario =  usuario
     self.livro = livro



class Controle:
  def __init__(self):
      # composição das entidades Livros, Usuarios e Emprestimos
      self.Livros = []
      self.Usuarios = []
      self.Emprestimos = []

  def CadastarUsuario(self, usuario):
    self.Usuarios.append(usuario)


  def CadastarLivro(self,livro):
    self.Livros.append(livro)


  def Emprestar(self, usuario, livro):
    self.Emprestimos.append(Emprestimo(usuario, livro))


  def Devolver(self, usuario, livro):
    for emprestimo in self.Emprestimos:
      if emprestimo.usuario == usuario and emprestimo.livro == livro:
        self.Emprestimos.remove(emprestimo)


  def ObterEmprestimos(self):
    return self.Emprestimos


  def ObterUsuario(self, nome):
    for usuario in self.Usuarios:
        if usuario.nome.casefold() == nome.strip().casefold():
            return usuario
    return None 


  def ObterLivro(self, titulo):
    for livro in self.Livros:
        if livro.titulo.casefold() == titulo.strip().casefold():
            return livro
    return None     




class BibliotecaBo:
  def __init__(self):
    self.Controle = Controle()


  def Run(self):
    self.LoopLivros()
    nome = self.LoopUsuarios()
    self.Executar(nome)


  def LoopUsuarios(self):
    numUsuario = 1  #int(input())
    nome = ""
    for i in range(numUsuario):
      usuario = Usuario(
          nome = input(),
          email = input())
      self.Controle.CadastarUsuario(usuario)
      nome = usuario.nome
    return nome


  def LoopLivros(self):
    numLivros = int(input())
    for i in range(numLivros):
      livro = Livro(
          titulo = input(),
          autor = input(),
          isbn = input())
      self.Controle.CadastarLivro(livro)


  def Executar(self, nome):
    usuario = self.Controle.ObterUsuario(nome)

    while True:
      operacao, titulo = self.LeOperacao()

      if operacao == "emprestar":
        livro = self.Controle.ObterLivro(titulo)
        if livro != None:
          self.Controle.Emprestar(usuario,livro)
          print("Livro emprestado: ", livro.titulo)

      elif operacao == "listar":
        emprestimos = self.Controle.ObterEmprestimos()
        if len(emprestimos) > 0:
          print(", ".join([emprestimo.livro.titulo for emprestimo in emprestimos]))

      elif operacao == "devolver":
        livro = self.Controle.ObterLivro(titulo)
        if livro != None:
          self.Controle.Devolver(usuario,livro)
          print("Livro devolvido: ", livro.titulo)

      elif operacao == "sair":
        return

      else:
        print("Operação inválida")


  def LeOperacao(self):
      operacao = input().strip().lower()

      if "emprestar: " in operacao:
        par = operacao.split(':')
        #operacao, titulo
        return ("emprestar", par[1].strip())


      elif "listar empréstimo" in operacao:
        return ("listar", None)


      elif "devolver: " in operacao:
        par = operacao.split(':')
        #operacao, titulo
        return ("devolver", par[1].strip())


      elif "sair" in operacao:
        return ("sair", None)

      else:
        return ("invalido", None)


def main():
  biblioteca = BibliotecaBo()
  biblioteca.Run()
  
if __name__ == "__main__":
  main()
